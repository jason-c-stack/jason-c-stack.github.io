GAP3 research prototype

k-W-Noncrossing Parking Functions

A compact computational implementation for building multichains in a noncrossing interval from dual-braid divisibility data and pairing those chains with reflection-subgroup cosets.

GAP3 complex reflection groups dual braid monoids parking functions

What the script does

The configured example starts with ComplexReflectionGroup(6). It obtains the simple left divisors of the dual Garside element and the atoms of the dual braid monoid.

  • Multichains(k,w) recursively builds length-k chains beginning at w.
  • NC_k(k) gathers chains over all simple elements.
  • Parking_k(k) pairs chains with left cosets of the subgroup generated by reflections below the first chain element.
  • The final loop prints distinct parking-function counts for successive chain lengths.

This is research code rather than a packaged library. It assumes a GAP3 environment providing the complex-reflection-group and dual-braid functions called in the first lines.

#k-W-noncrossing parking function code
W:= ComplexReflectionGroup(6);
B:=DualBraidMonoid(W);
D:=DualBraid(W);
S:=LeftDivisorsSimple(B,B.delta);
A:=B.atoms;

#Gives all multichains begining with w
Multichains:=function(k,w)
    local MCw,L,MCu,R,u,M;
    MCw:=[];
    if k>1 then
        for R in S do 
            for u in R do
                if LeftGcd(D(w),D(u))[1]=D(w) then 
                    MCu:=Multichains(k-1,u);
                    for M in MCu do 
                        L:=[w];
                        Append(L,M);
                        if not(L in MCw) then 
                        Add(MCw,L);
                        fi;
                    od;
                fi;
            od;
        od;
    fi;

    if k=1 then
        MCw:=[[w]];
    fi;

    return MCw;
end;

#Gives all multichains begining with w
Mchains:=function(k,w)
	local i, MC, M, pi, R, u;
	i:=1;
	while i < k do 
		MC:= [[w]];
		for M in MC do 
			for R in S do 
				for u in R do
					pi:=M[Size(Set(M))];
					if LeftGcd(D(pi),D(u))[1]=D(pi) then 
						newM:=
						Add(MC,u);
					fi;
				od;
			od;
		od;
	i:=i+1;
	od;

	return MC;
end;

#Uses Multichains function to give all k-multichains 
NC_k:=function(k)
	local NC,R,u,MCu;
	NC:=[];
	for R in S do
		for u in R do
			MCu:=Multichains(k,u);
			Add(NC,MCu);
		od;
	od;
	return NC;
end;


#Gives all k parking functions as a list of entries [wW_x,x,x_1,...,x_{k-1}] of cosets and k-chains

Parking_k:=function(k)
	local P,R,u,MC,TNC,t,G,Wu,LC,L,c,Lc;
	P:=[];
	for R in S do
		for u in R do
			MC:=Multichains(k,u);
			TNC:=[]; #List the reflections below u 
			for t in A do 
				G:=LeftGcd(D(t),D(u));
				if G[1]=D(t) then
					Add(TNC,t);
				fi;
			od;
			Wu:=Subgroup(W,TNC);
			LC:=LeftCosets(W,Wu);
			for L in LC do
				for c in MC do
					Lc:=[L];
					Append(Lc,c);
					Add(P,Lc);
				od;
			od;
		od;
	od;
	return P;
end;

count:=0;
for R in S do 
	for j in R do 
		count:=count+1;
	od;
od;

for k in [1..count] do 
	Print(Size(Set(Parking_k(k))),"\n",k,"\n");
od;
